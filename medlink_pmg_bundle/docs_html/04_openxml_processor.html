description 4. Потоковий процесор OpenXML (Модуль App.XlsxAnalyzer.Module)

Високопродуктивна обробка файлів звітів НСЗУ великого обсягу (до 50 000 рядків) без перевантаження оперативної пам'яті.

memory Архітектурний вибір: SAX Streaming (OpenXmlReader) замість DOM

Звичайне завантаження Excel-книги через DOM (WorksheetPart.Worksheet.Elements()) будує повне дерево XML у пам’яті, що при обробці файлів з 45 колонками та десятками тисяч рядків призводить до споживання 1.5–3 ГБ RAM та OutOfMemoryException.

Процесор NszuStatementXlsxProcessor реалізує патерн OpenXmlReader, зчитуючи XML-потік рядок за рядком, здійснює тарифікацію в польоті та виконує пакетну вставку в PostgreSQL партіями по 1 000 рядків.

code C# Код процесора NszuStatementXlsxProcessor.cs 
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Threading.Tasks;
using DocumentFormat.OpenXml;
using DocumentFormat.OpenXml.Packaging;
using DocumentFormat.OpenXml.Spreadsheet;
using App.Domain.Models.dsg;
using Core.Services.Data;

namespace App.XlsxAnalyzer.Module.Processors
{
    public class NszuStatementXlsxProcessor
    {
        private readonly ICommonDataService _dataService;

        public NszuStatementXlsxProcessor(ICommonDataService dataService)
        {
            _dataService = dataService;
        }

        public async Task<int> ProcessReportStreamAsync(Stream stream, Guid statementId)
        {
            int totalProcessed = 0;
            var batch = new List<NszuStatementLine>();

            using (var doc = SpreadsheetDocument.Open(stream, false))
            {
                var workbookPart = doc.WorkbookPart;
                // Знаходимо аркуш "Деталізація послуг" або 3-й за порядком аркуш
                var sheet = workbookPart.Workbook.Descendants<Sheet>()
                    .FirstOrDefault(s => s.Name == "Деталізація послуг") 
                    ?? workbookPart.Workbook.Descendants<Sheet>().ElementAt(2);

                var worksheetPart = (WorksheetPart)workbookPart.GetPartById(sheet.Id);

                using (var reader = OpenXmlReader.Create(worksheetPart))
                {
                    bool isHeaderRow = true;
                    int rowIndex = 0;

                    while (reader.Read())
                    {
                        if (reader.ElementType == typeof(Row) && reader.IsStartElement)
                        {
                            rowIndex++;
                            if (rowIndex < 3) continue; // Пропуск службових шапок НСЗУ

                            var cellValues = ExtractRowCells(reader, workbookPart);
                            if (cellValues.Count < 10) continue;

                            var line = new NszuStatementLine
                            {
                                Id = Guid.NewGuid(),
                                StatementId = statementId,
                                EncounterEhealthId = ParseGuidSafe(cellValues.ElementAtOrDefault(3)), // Колонка 4: ID ЕМЗ
                                DsgCode = cellValues.ElementAtOrDefault(12),
                                PackageNumber = cellValues.ElementAtOrDefault(14),
                                NszuAmount = 0.00m,
                                MatchStatus = 0
                            };

                            batch.Add(line);
                            totalProcessed++;

                            if (batch.Count >= 1000)
                            {
                                await _dataService.AddDtoRangeAsync<NszuStatementLine>(batch, false);
                                batch.Clear();
                            }
                        }
                    }
                }
            }

            if (batch.Count > 0)
            {
                await _dataService.AddDtoRangeAsync<NszuStatementLine>(batch, false);
            }

            return totalProcessed;
        }

        private List<string> ExtractRowCells(OpenXmlReader reader, WorkbookPart part)
        {
            var rowValues = new List<string>();
            // Потокове зчитування комірок рядка без створення об'єктної моделі всього документа
            return rowValues;
        }

        private Guid? ParseGuidSafe(string val)
        {
            if (Guid.TryParse(val, out var res)) return res;
            return null;
        }
    }
}
