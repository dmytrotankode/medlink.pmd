storage 1. Специфікація схеми бази даних (PostgreSQL) та Entity Framework Core

Детальний опис таблиць зберігання звітів НСЗУ, результатів аналітики ПМГ-2026 та зв'язків із ядрами клінічних сутностей МІС evomis.

verified Результати аудиту тестової бази даних evomis-test (PostgreSQL 14)

Підключення до тестової БД через VPN (192.168.255.1:5432) підтвердило наявність сформованого ядра таблиць для ДСГ:

  • dsg_nszu_statement — заголовки імпортованих звітних файлів НСЗУ.
  • dsg_nszu_statement_line — рядки деталізації послуг зі звіту НСЗУ (аркуш «Деталізація послуг»).
  • dsg_analysis_result — результати розрахунку тарифів та перевірки правил валідації (із полями calculation (jsonb) та findings (jsonb)).
  • dsg_tariff_setting — налаштування тарифів ПМГ-2026: Базова ставка 8 735,00 ₴, частка глобальної ставки 0,55.
  • dsg_group_weight — 471 запис вагових коефіцієнтів ДСГ (наприклад, A40: 12.7030, B01: 2.9420).
  • dsg_rule_config — 58 нормативних правил перевірки (FR-01, FR-02, PK-01A, PK-04, тощо).

using System;
using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;
using Core.Base.Data;

namespace App.Domain.Models.dsg
{
    [Table("dsg_nszu_statement")]
    public class NszuStatement : CoreEntity
    {
        [Column("organization_id")]
        public Guid OrganizationId { get; set; }

        [Column("period_from")]
        public DateTime PeriodFrom { get; set; }

        [Column("period_to")]
        public DateTime PeriodTo { get; set; }

        [Column("file_name")]
        [MaxLength(500)]
        public string FileName { get; set; }

        [Column("imported_at")]
        public DateTime ImportedAt { get; set; } = DateTime.UtcNow;

        [Column("imported_by")]
        public Guid? ImportedBy { get; set; }
    }

    [Table("dsg_nszu_statement_line")]
    public class NszuStatementLine : CoreEntity
    {
        [Column("statement_id")]
        public Guid StatementId { get; set; }

        [Column("encounter_ehealth_id")]
        public Guid? EncounterEhealthId { get; set; }

        [Column("matched_encounter_id")]
        public Guid? MatchedEncounterId { get; set; }

        [Column("dsg_code")]
        [MaxLength(64)]
        public string DsgCode { get; set; }

        [Column("package_number")]
        [MaxLength(32)]
        public string PackageNumber { get; set; }

        [Column("nszu_amount")]
        public decimal NszuAmount { get; set; }

        [Column("mis_amount")]
        public decimal? MisAmount { get; set; }

        [Column("difference")]
        public decimal? Difference { get; set; }

        [Column("match_status")]
        public int MatchStatus { get; set; }

        [ForeignKey("StatementId")]
        public virtual NszuStatement Statement { get; set; }
    }

    [Table("dsg_analysis_result")]
    public class DsgAnalysisResult : CoreEntity
    {
        [Column("encounter_id")]
        public Guid EncounterId { get; set; }

        [Column("organization_id")]
        public Guid OrganizationId { get; set; }

        [Column("package_number")]
        [MaxLength(16)]
        public string PackageNumber { get; set; }

        [Column("dsg_group_code")]
        [MaxLength(32)]
        public string DsgGroupCode { get; set; }

        [Column("status")]
        public int Status { get; set; }

        [Column("tariff")]
        public decimal Tariff { get; set; }

        [Column("estimated_payment")]
        public decimal EstimatedPayment { get; set; }

        [Column("weight_coef")]
        public decimal WeightCoef { get; set; }

        [Column("calculation", TypeName = "jsonb")]
        public string CalculationJson { get; set; }

        [Column("findings", TypeName = "jsonb")]
        public string FindingsJson { get; set; }

        [Column("primary_icd10_code")]
        [MaxLength(32)]
        public string PrimaryIcd10Code { get; set; }

        [Column("employee_id")]
        public Guid? EmployeeId { get; set; }

        [Column("department_id")]
        public Guid? DepartmentId { get; set; }
    }
}

        .HasOne(x => x.Statement)
        .WithMany()
        .HasForeignKey(x => x.StatementId)
        .OnDelete(DeleteBehavior.Cascade);

    // Індекси для швидкої 2-Way звірки за Encounter.EhealthId
